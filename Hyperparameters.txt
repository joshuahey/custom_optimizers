# ============================================================
# DG / DG-Delta hyperparameters
# ============================================================
#
# Main parameters
# ---------------
# h : float, needs tuning 
#     Step size. 
#     Larger h may speed things up, but if it is too large, the fixed-point iteration may become unstable or fail to converge.
#
# gamma : float (default choices 0.5 - 0.9, for DG
#               (default choices 0.1 - 0.5, for DG - delta)
#     Damping parameter (classical momentum damping constant)
#     
# delta : float, needs tuning   (DG_Delta only) 
#     Extra gradient damping in the theta update.
#     If it is too large, it may slow convergence.
#     Small positive values are usually used.
# 
# 
# Default parameters
# ---------------
# fp_iters : int (default choices 1 - 3)
#     Maximum number of fixed-point iterations used to approximate theta_{k+1}.
#     Since DG and DG-Delta are implicit methods, theta_{k+1} is obtained by
#     solving a fixed-point equation.
#
# avf_samples : int (default choices 1 or 2)
#     Number of quadrature samples used to approximate the AVF discrete gradient
#
#         gbar = ∫_0^1 ∇L((1-s) theta_k + s theta_{k+1}) ds.
#
#     Larger values give a more accurate approximation, but cost increases.
#
# fp_tol : float (default choices 1e-1 -  1e-3)
#     Tolerance for early stopping of the fixed-point iteration.
#     If successive fixed-point iterates are close enough, the inner solve stops
#     early. Smaller fp_tol gives a more accurate inner solve, but costs more.



# ============================================================
# SIDG / SIDG-Delta hyperparameters
# ============================================================
#
# Main parameters: 
# ---------------
# h : float, needs tuning 
#     Step size.
#     Larger values may accelerate convergence, but if chosen too large, they can reduce the stability of the semi-implicit update.
#
# gamma : float (default choices 0.5 - 0.9, for SIDG
#               (default choices 0.1 - 0.5, for SIDG - delta)
#     Damping parameter (classical momentum damping constant)
#
# delta : float, needs tuning  (SIDG_Delta only)
#     Additional damping parameter in the theta-update.
#     Small positive values are usually used. 
#
# history_size : int, needs tuning 
#     Number of L-BFGS curvature pairs stored.
#     Larger values may improve the inverse-Hessian approximation, but increase memory.
#
# cg_max_iter : int (default choices 5 or 25)
#     Maximum number of CG iterations for the inner linear solves.

# Default parameters
# ---------------
# init_H_scale : float (default = 1.0)
#     Initial matrix scaling when no curvature information is present.
#
# cg_tol : float  (default choices 1e-6)
#     Tolerance for the conjugate-gradient inner solve.
#     Smaller values give a more accurate inner solve, but cost more.
#
# curvature_eps : float (default = 1e-8)
#     Curvature threshold used when deciding whether to accept an L-BFGS pair.
#     Pairs with very small or nonpositive curvature are discarded.
